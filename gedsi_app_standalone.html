/** * GEDSI Activity Participation Recorder — Google Apps Script backend (JSONP) * * ວິທີໃຊ້: * 1. ເປີດ Google Sheet ໃໝ່, ຕັ້ງຊື່ຊີດວ່າ "Data" * 2. ໃສ່ header ແຖວທຳອິດ: * id | ຊື່ກິດຈະກຳ | ວັນທີກິດຈະກຳ | ແຂວງ | ເມືອງ | ຜູ້ໃຫ້ທຶນ | ຍິງ | ຊາຍ | ອື່ນໆ | ອາຍຸ<18 | ອາຍຸ18-35 | ອາຍຸ35-60 | ອາຍຸ>60 | * ມີຄວາມພິການ | ດ້ານຮ່າງກາຍ | ດ້ານປະສາດສຳພັດ | ດ້ານສັງຄົມຈິດຕະວິທະຍາ | ດ້ານສະໝອງແລະສະຕິປັນຍາ | * ສະຫຼຸບ | ເວລາບັນທຶກ * 3. Deploy → Manage deployments → ✏️ ແກ້ໄຂ → Version: New → Deploy */ function getSheet_() { var ss = SpreadsheetApp.getActiveSpreadsheet(); return ss.getSheetByName('Data') || ss.getActiveSheet(); } function doGet(e) { var p = (e && e.parameter) || {}; var action = p.action || 'list'; var sheet = getSheet_(); var result; if (action === 'save') result = handleSave_(sheet, p); else if (action === 'delete') result = handleDelete_(sheet, p.id); else result = listAll_(sheet); if (p.callback) { return ContentService .createTextOutput(p.callback + '(' + JSON.stringify(result) + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return jsonOut_(result); } function handleSave_(sheet, p) { try { var id = p.id || String(new Date().getTime()); sheet.appendRow([ id, p.activityName || '', p.activityDate || '', p.province || '', p.district || '', p.funder || '', p.female || 0, p.male || 0, p.otherGender || 0, p.ageU18 || 0, p.age18_35 || 0, p.age35_60 || 0, p.age60plus || 0, p.hasDisability || 'No', p.disPhysical || 0, p.disSensory || 0, p.disPsychosocial || 0, p.disIntellectual || 0, p.summary || '', new Date() ]); return { status: 'ok', id: id }; } catch (err) { return { status: 'error', message: err.message }; } } function listAll_(sheet) { try { var values = sheet.getDataRange().getValues(); var entries = []; for (var i = 1; i < values.length; i++) { var r = values[i]; if (!r[0]) continue; entries.push({ id: String(r[0]), activityName: r[1] || '', activityDate: formatDate_(r[2]), province: r[3] || '', district: r[4] || '', funder: r[5] || '', female: Number(r[6]) || 0, male: Number(r[7]) || 0, otherGender: Number(r[8]) || 0, ageU18: Number(r[9]) || 0, age18_35: Number(r[10]) || 0, age35_60: Number(r[11]) || 0, age60plus: Number(r[12]) || 0, hasDisability: r[13] || 'No', disPhysical: Number(r[14]) || 0, disSensory: Number(r[15]) || 0, disPsychosocial: Number(r[16]) || 0, disIntellectual: Number(r[17]) || 0, summary: r[18] || '' }); } return { status: 'ok', entries: entries }; } catch (err) { return { status: 'error', message: err.message, entries: [] }; } } function formatDate_(v) { if (!v) return ''; if (Object.prototype.toString.call(v) === '[object Date]') { var y = v.getFullYear(); var m = ('0' + (v.getMonth() + 1)).slice(-2); var d = ('0' + v.getDate()).slice(-2); return y + '-' + m + '-' + d; } return String(v); } function handleDelete_(sheet, id) { try { var values = sheet.getDataRange().getValues(); for (var i = 1; i < values.length; i++) { if (String(values[i][0]) === String(id)) { sheet.deleteRow(i + 1); break; } } return { status: 'ok' }; } catch (err) { return { status: 'error', message: err.message }; } } function jsonOut_(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
